1	// Assumption: sampling uniformly from a set S is indistinguishable from
2	// sampling from S \ bookkeeping.
3	
4	// Justification: FrogLang's <-uniq[bookkeeping] semantics implicitly add
5	// the sampled value to bookkeeping, so over q queries to Samp the
6	// NoReplacement game is guaranteed to return q distinct values. The two
7	// games are then identical conditioned on no collision occurring in the
8	// Replacement game, since NoReplacement differs from Replacement only by
9	// rejecting values already in bookkeeping. The birthday-paradox collision
10	// probability over q independent uniform samples from S is at most
11	// q(q-1)/(2|S|), so the statistical distance -- and hence the
12	// distinguishing advantage -- is bounded by at most q^2 / |S|. (This
13	// bound is meaningful only when q << |S|^(1/2); at q \approx |S|^(1/2)
14	// it is trivial.)
15

17	    S Samp(Set<S> bookkeeping) {
18	        S val <- S;
